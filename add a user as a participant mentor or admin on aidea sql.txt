DECLARE @db SYSNAME = DB_NAME();
IF @db <> N'aidea'
   OR OBJECT_ID(N'dbo.schema_migrations', N'U') IS NULL
   OR OBJECT_ID(N'dbo.user_roles', N'U') IS NULL
   OR NOT EXISTS (SELECT 1 FROM dbo.schema_migrations WHERE name = N'0001_schema.sql')

  RAISERROR(N'Not the AIdea database (connected to %s). Nothing was changed.', 16, 1, @db);
  SET NOEXEC ON;

SET XACT_ABORT ON;

DECLARE @people TABLE (
  email     NVARCHAR(320) NOT NULL PRIMARY KEY,
  full_name NVARCHAR(200) NULL,   -- optional; replaced by the AD name at sign-in
  tier      NVARCHAR(40)  NOT NULL CHECK (tier IN ('participant', 'mentor', 'admin'))
);

BEGIN TRAN;

-- 1. Account (no password: AD is the only way in)
INSERT INTO users (email)
SELECT p.email FROM @people p
 WHERE NOT EXISTS (SELECT 1 FROM users u WHERE u.email = p.email);

-- 2. Profile
INSERT INTO profiles (id, email, full_name)
SELECT u.id, u.email, COALESCE(p.full_name, p.email)
  FROM @people p JOIN users u ON u.email = p.email
 WHERE NOT EXISTS (SELECT 1 FROM profiles pr WHERE pr.id = u.id);

-- 3. Remove any current role (each person has exactly one)
DELETE ur
  FROM user_roles ur
  JOIN roles r   ON r.id = ur.role_id
  JOIN users u   ON u.id = ur.user_id
  JOIN @people p ON p.email = u.email
 WHERE r.name IN ('participant', 'mentor', 'admin', 'developer');

-- 4. Give the requested role
INSERT INTO user_roles (user_id, role_id)
SELECT u.id, r.id
  FROM @people p
  JOIN users u ON u.email = p.email
  JOIN roles r ON r.name = p.tier;

-- 5. Everyone can vote
INSERT INTO user_roles (user_id, role_id)
SELECT u.id, r.id
  FROM @people p
  JOIN users u ON u.email = p.email
  JOIN roles r ON r.name = 'employee_voter'
 WHERE NOT EXISTS (SELECT 1 FROM user_roles ur WHERE ur.user_id = u.id AND ur.role_id = r.id);

-- 6. Mentors need a mentor profile
INSERT INTO mentor_profiles (profile_id)
SELECT u.id
  FROM @people p JOIN users u ON u.email = p.email
 WHERE p.tier = 'mentor'
   AND NOT EXISTS (SELECT 1 FROM mentor_profiles m WHERE m.profile_id = u.id);

COMMIT;

-- Review
SELECT u.email, pr.full_name, STRING_AGG(r.name, ', ') AS roles,
       CASE WHEN u.ad_object_guid IS NULL THEN 'not signed in yet' ELSE 'signed in' END AS status
  FROM @people p
  JOIN users u       ON u.email = p.email
  JOIN profiles pr   ON pr.id = u.id
  JOIN user_roles ur ON ur.user_id = u.id
  JOIN roles r       ON r.id = ur.role_id
 GROUP BY u.email, pr.full_name, u.ad_object_guid
 ORDER BY u.email;
GO
SET NOEXEC OFF;
